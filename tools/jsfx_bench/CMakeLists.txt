cmake_minimum_required(VERSION 3.22)
project(ysfxbench CXX C ASM)
set(CMAKE_CXX_STANDARD 17)
set(YSFX_PORTABLE OFF CACHE BOOL "" FORCE)
set(YSFX_GFX OFF CACHE BOOL "" FORCE)
set(YSFX_PLUGIN OFF CACHE BOOL "" FORCE)
set(YSFX_TESTS OFF CACHE BOOL "" FORCE)
set(YSFX_TOOLS OFF CACHE BOOL "" FORCE)
set(YSFX_SKIP_CHECKSUM ON CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../3rd_party/ysfx ysfx)
foreach(t ysfx-private lice wdl-base eel2)
  if(TARGET ${t})
    target_compile_options(${t} PRIVATE "-include${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp/jsfx/ysfx_android_compat.h")
  endif()
endforeach()
add_executable(bench bench.cpp)
target_link_libraries(bench PRIVATE ysfx)
target_sources(bench PRIVATE prealloc.cpp)
target_include_directories(bench PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/../../3rd_party/ysfx/sources
  ${CMAKE_CURRENT_SOURCE_DIR}/../../3rd_party/ysfx/thirdparty/WDL/source)
target_compile_options(bench PRIVATE "-include${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp/jsfx/ysfx_android_compat.h")
target_compile_options(bench PRIVATE -fsigned-char)

# standalone micro-benchmarks (no ysfx dependency)
add_executable(sliders sliders.cpp)
add_executable(fpcr fpcr.c)
